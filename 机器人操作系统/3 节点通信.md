### 话题通信编程
- 创建功能包
  `ros2 pkg create --build-type ament_python --license Apache-2.0 py_demo_topic --dependencies rclpy std_msgs`
  --dependencies中是依赖包的包名，空格分隔
- 编写发布者节点
```python
# 导入ROS2客户端库
import rclpy
# 导入Node基类，所有节点都需要继承此类
from rclpy.node import Node
# 导入标准消息类型String
from std_msgs.msg import String

class DemoPublisher(Node):
    """发布者节点类，用于定期发布消息到指定话题"""
    
    def __init__(self):
        """
        节点初始化函数
        - 创建节点名为'my_publisher'
        - 创建发布者对象，话题名为'chd_topic'，队列大小为10
        - 创建定时器，每0.5秒触发一次回调函数
        """
        # 调用父类构造函数，设置节点名
        super().__init__('my_publisher')
        
        # 创建发布者对象：
        # - String: 消息类型（标准字符串消息）
        # - 'chd_topic': 话题名称（其他节点通过此名称订阅）
        # - 10: 队列大小（缓存的消息数量）
        self.publisher_chd = self.create_publisher(String, 'chd_topic', 10)
        
        # 设置定时器周期为0.5秒
        timer_period = 0.5  # seconds
        
        # 创建定时器：
        # - timer_period: 定时器周期（秒）
        # - self.timer_callback: 定时器到期时的回调函数
        self.timer_chd = self.create_timer(timer_period, self.timer_callback)
        
        # 初始化消息索引计数器
        self.index = 0
        
        # 在日志中输出初始化完成信息
        self.get_logger().info('DemoPublisher节点初始化完成，开始发布消息...')

    def timer_callback(self):
        """
        定时器回调函数
        - 每0.5秒自动执行一次
        - 创建并发布消息到'chd_topic'话题
        - 在日志中记录发布的消息内容
        """
        # 创建String类型的消息对象
        msg = String()
        
        # 设置消息内容：包含问候语和当前索引
        msg.data = f'Hello World and CHD: index={self.index}'
        
        # 发布消息到'chd_topic'话题
        self.publisher_chd.publish(msg)
        
        # 在日志中记录发布的消息内容（INFO级别）
        self.get_logger().info(f'Publishing: "{msg.data}"')
        
        # 索引自增，为下一条消息准备
        self.index += 1

def main(args=None):
    """
    主函数：节点的主要执行流程
    """
    # 初始化ROS2客户端库，必须首先调用
    rclpy.init(args=args)
    
    try:
        # 创建发布者节点实例
        demo_publisher = DemoPublisher()
        
        # 进入ROS2事件循环，保持节点运行并处理回调
        # spin()会阻塞直到节点被关闭
        rclpy.spin(demo_publisher)
        
    except KeyboardInterrupt:
        # 捕获Ctrl+C中断信号
        demo_publisher.get_logger().info('收到中断信号，节点正在关闭...')
    except Exception as e:
        # 捕获其他异常
        demo_publisher.get_logger().error(f'节点运行出错: {e}')
    finally:
        # 清理资源：销毁节点
        if 'demo_publisher' in locals():
            demo_publisher.destroy_node()
        
        # 关闭ROS2客户端库
        rclpy.shutdown()

# Python标准的主程序入口
if __name__ == '__main__':
    main()
```

```python
# 导入ROS2客户端库
import rclpy
# 导入Node基类，所有节点都需要继承此类
from rclpy.node import Node
# 导入标准消息类型String
from std_msgs.msg import String

class DemoSubscriber(Node):
    """订阅者节点类，用于订阅指定话题并接收消息"""
    
    def __init__(self):
        """
        节点初始化函数
        - 创建节点名为'my_subscriber'
        - 创建订阅者对象，订阅'chd_topic'话题
        - 设置消息接收时的回调函数
        """
        # 调用父类构造函数，设置节点名
        super().__init__('my_subscriber')
        
        # 创建订阅者对象：
        # - String: 订阅的消息类型（必须与发布者发送的类型匹配）
        # - 'chd_topic': 要订阅的话题名称
        # - self.listener_callback: 消息到达时的回调函数
        # - 10: 队列大小（缓存的消息数量）
        self.subscription_chd = self.create_subscription(
            String,
            'chd_topic',
            self.listener_callback,
            10)
        
        # 这行代码是为了防止未使用变量警告（PEP8规范）
        # 在实际功能上不是必需的，但可以保持代码整洁
        self.subscription_chd  # prevent unused variable warning
        
        # 在日志中输出初始化完成信息
        self.get_logger().info('DemoSubscriber节点初始化完成，开始监听chd_topic话题...')

    def listener_callback(self, msg):
        """
        消息接收回调函数
        - 当有消息发布到'chd_topic'话题时自动调用
        - msg: 接收到的消息对象
        
        :param msg: 接收到的String类型消息
        """
        # 在日志中记录接收到的消息内容
        # 使用f-string格式化输出，显示消息内容
        self.get_logger().info(f'I heard: "{msg.data}"')

def main(args=None):
    """
    主函数：节点的主要执行流程
    """
    # 初始化ROS2客户端库，必须首先调用
    rclpy.init(args=args)
    
    try:
        # 创建订阅者节点实例
        demo_subscriber = DemoSubscriber()
        
        # 在日志中输出节点启动信息
        demo_subscriber.get_logger().info('订阅者节点启动，进入消息监听状态...')
        
        # 进入ROS2事件循环，保持节点运行并处理回调
        # spin()会阻塞直到节点被关闭，期间会监听到来的消息
        rclpy.spin(demo_subscriber)
        
    except KeyboardInterrupt:
        # 捕获Ctrl+C中断信号
        demo_subscriber.get_logger().info('收到中断信号，节点正在关闭...')
    except Exception as e:
        # 捕获其他异常
        demo_subscriber.get_logger().error(f'节点运行出错: {e}')
    finally:
        # 清理资源：销毁节点
        # 修正：这里应该是demo_subscriber，不是minimal_subscriber
        if 'demo_subscriber' in locals():
            demo_subscriber.destroy_node()
            demo_subscriber.get_logger().info('节点资源已清理')
        
        # 关闭ROS2客户端库
        rclpy.shutdown()

# Python标准的主程序入口
if __name__ == '__main__':
    main()
```

## 节点的命名空间、重命名
### 命名空间namespace
将命名分组储存，便于管理，允许不同命名空间中存在重名。

默认的命名空间是根目录，即'/'

命名空间路径的重映射方法：
`ros2 run <package-name>  <executable>  --ros-args --remap __ns:=<new-ns>`
\<executable\>填要被重命名的空间名
\_\_ns代表这里重命名的是namespace，下面的__node等同理
### 节点重命名
`ros2 run <package-name>  <executable>  --ros-args --remap __node:=<new-node-name>`

如果要同时重命名命名空间和节点：
`ros2 run <package-name>  <executable>  --ros-args --remap __ns:=<new-ns>--remap __node:=<new-node-name>`
### 话题重命名
`ros2 run <package-name>  <executable>  --ros-args --remap <old-topic-name>:=<new-topic-name>`
## 作业
> 话题通信程序中，订阅者的回调函数作用是什么？该函数和spin函数有什么关系？请描述大概的处理过程。

订阅者的回调函数用于处理接收到的话题消息。spin函数是一个阻塞函数，不断轮询到达节点的事件，一旦发现有所订阅的消息到来就会通过线程去调用回调函数，实现消息的处理。

> 为了通过话题通信方式将机器人的位置和环境温湿度信息发布到系统中（位置用x、y、z表示，温度用t表示、湿度用h表示），要求用自定义接口实现该话题类型，应该如何定义该接口文件？在python模块中如何导入该类型以便使用?

可以创建一个C++功能包my_interface实现接口类型。比如在msg文件夹中创建Condition.msg文件，内容包括：
- float 64 x
- float 64 y
- float 64 z
- float 32 t
- float 32 h

通过构建该功能包，生成对应的C++和python接口类型Condition
在python模块中可使用from my_interface.msg
import Condition即可导入该类型，类型名为Condition(参考上面正文的自定义话题通信)

> 为了只针对特定功能包进行构建，可以使用哪些构建选项，这些选项有什么区别？

可采用构建选项：--packages-select或--packages-up-to进行构建，前者只构建指定包，不会处理其依赖的包；后者会递归构建所有依赖，将所有相关的功能包同步更新。
如`colcon build --packages-select <package_name>`

> 构建选项中的—symlink-install的作用是什么？

该选项能够在安装目录通过符号链接的方式指向二进制文件和资源文件，减少了默认没有该选项产生的文件I/O操作，更重要的是符号链接直接指向源码文件，更改python源码和资源文件无需重新构建，提升了调试效率。
